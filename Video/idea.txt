        let originalImage = null;
        let slicedImages = []; // 存储切好的图片数据 Blob
        let originalFileName = "image";

        // 监听文件上传
        document.getElementById('fileInput').addEventListener('change', function(e) {
            const file = e.target.files[0];
            if (!file) return;
            
            originalFileName = file.name.split('.')[0];
            const reader = new FileReader();
            
            reader.onload = function(event) {
                const img = new Image();
                img.onload = function() {
                    originalImage = img;
                    document.getElementById('statusMsg').innerText = `图片加载成功: ${img.width}x${img.height}px`;
                    // 自动触发一次预览（可选）
                    // processImage();
                }
                img.src = event.target.result;
            }
            reader.readAsDataURL(file);
        });

        function processImage() {
            if (!originalImage) {
                alert("请先选择一张图片！");
                return;
            }

            const rows = parseInt(document.getElementById('rowsInput').value);
            const cols = parseInt(document.getElementById('colsInput').value);
            
            if (rows < 1 || cols < 1) {
                alert("行数和列数必须大于0");
                return;
            }

            const previewArea = document.getElementById('preview-area');
            previewArea.innerHTML = ''; // 清空
            slicedImages = []; // 重置存储

            // 创建网格容器
            const gridContainer = document.createElement('div');
            gridContainer.className = 'grid-display';
            gridContainer.style.gridTemplateColumns = `repeat(${cols}, 1fr)`;
            // 限制预览图最大宽度，防止撑爆页面
            gridContainer.style.maxWidth = '100%'; 
            
            // 计算每块的尺寸
            const itemWidth = Math.floor(originalImage.width / cols);
            const itemHeight = Math.floor(originalImage.height / rows);

            // 遍历切割
            for (let r = 0; r < rows; r++) {
                for (let c = 0; c < cols; c++) {
                    const canvas = document.createElement('canvas');
                    canvas.width = itemWidth;
                    canvas.height = itemHeight;
                    const ctx = canvas.getContext('2d');

                    // drawImage(image, sx, sy, sWidth, sHeight, dx, dy, dWidth, dHeight)
                    ctx.drawImage(
                        originalImage, 
                        c * itemWidth, r * itemHeight, itemWidth, itemHeight, // 源图坐标和尺寸
                        0, 0, itemWidth, itemHeight // 目标画布坐标和尺寸
                    );

                    // 1. 用于显示的 img 标签
                    const imgTag = document.createElement('img');
                    imgTag.src = canvas.toDataURL('image/png');
                    imgTag.className = 'grid-item';
                    imgTag.title = `第${r+1}行, 第${c+1}列 (点击下载这张)`;
                    
                    // 单击单独下载
                    (function(row, col, dataUrl) {
                        imgTag.onclick = () => {
                            const link = document.createElement('a');
                            link.download = `${originalFileName}_${row+1}_${col+1}.png`;
                            link.href = dataUrl;
                            link.click();
                        };
                    })(r, c, canvas.toDataURL('image/png'));

                    gridContainer.appendChild(imgTag);

                    // 2. 将数据存入 slicedImages 数组用于打包下载
                    // canvas 转 blob
                    canvas.toBlob((blob) => {
                        slicedImages.push({
                            name: `${originalFileName}_${r+1}_${c+1}.png`,
                            blob: blob
                        });
                        
                        // 检查是否所有图片都处理完毕
                        if (slicedImages.length === rows * cols) {
                             document.getElementById('downloadBtn').classList.remove('hidden');
                             document.getElementById('statusMsg').innerText = `切割完成！共 ${rows*cols} 张图片。`;
                        }
                    }, 'image/png');
                }
            }
            
            previewArea.appendChild(gridContainer);
        }

        function downloadZip() {
            if (slicedImages.length === 0) return;

            const zip = new JSZip();
            const folder = zip.folder("split_images");

            slicedImages.forEach(item => {
                folder.file(item.name, item.blob);
            });

            document.getElementById('downloadBtn').innerText = "正在打包...";
            
            zip.generateAsync({type:"blob"})
            .then(function(content) {
                saveAs(content, `${originalFileName}_split.zip`);
                document.getElementById('downloadBtn').innerText = "📦 打包下载所有切片";
            });
        }

画面流畅

前期准备
所有生成过比较长的有剧情的ai视频的佬友们可能都经历过，ai生成的视频每一段流畅性都没问题，但是一旦拼接到一起，段与段的衔接的就很生硬，其实就是ai没有长视频的分镜概念。
首尾帧一定程度可以解决问题，但是sora没有首尾帧，所以前期准备上不管是文字脚本还是静态分镜一定要注意确保各个视频生成后的逻辑连贯性。
虽然提供的脚本制作关键词里提到了确保视觉流畅且逻辑连贯，还需要人工审核一遍

后期剪辑
没什么好说的，sora视频最短都有10秒，剪掉不需要的，视频之间加一些叠加，闪黑之类的转场特效

音乐流畅
sora每一段视频配音（背景音乐/效果音）都不同，从这一段到下一段音乐风格会突然变化，可以调节每一段音频的淡入淡出，或者保留原音乐下添加一段完整的背景音从头贯穿到尾。
这些都不是什么好办法，期待有佬友能给出别的解决方案
