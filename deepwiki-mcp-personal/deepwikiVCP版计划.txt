2. 插件本体实现 (deepwiki-processor.js)
基于现有的核心功能模块，实现VCP标准的stdin/stdout通信：

// 核心模式实现  
const fs = require('fs').promises;  
const path = require('path');  
  
// 复用现有模块  
const { crawl } = require('./lib/httpCrawler');  
const { htmlToMarkdown } = require('./converter/htmlToMarkdown');  
const { FetchRequest } = require('./schemas/deepwiki');  
  
async function main() {  
  try {  
    // 1. 读取stdin  
    const input = await readStdin();  
    const request = JSON.parse(input);  
      
    // 2. 命令分发  
    let result;  
    switch (request.command) {  
      case 'FetchContent':  
        result = await handleFetchContent(request);  
        break;  
      case 'SearchContent':  
        result = await handleSearchContent(request);  
        break;  
      default:  
        throw new Error(`Unknown command: ${request.command}`);  
    }  
      
    // 3. 输出结果  
    console.log(JSON.stringify({  
      status: "success",  
      result: result  
    }));  
      
  } catch (error) {  
    console.log(JSON.stringify({  
      status: "error",  
      error: error.message  
    }));  
  }  
    
  process.exit(0);  
}
3. 核心功能重构
FetchContent命令实现：

async function handleFetchContent(request) {  
  // 复用现有URL规范化逻辑  
  const normalizedUrl = await normalizeUrl(request.url);  
    
  // 复用现有爬取逻辑  
  const crawlResult = await crawl({  
    root: new URL(normalizedUrl),  
    maxDepth: request.maxDepth || 1,  
    emit: () => {},  
    verbose: false  
  });  
    
  // 复用现有转换逻辑  
  const pages = await Promise.all(  
    Object.entries(crawlResult.html).map(async ([path, html]) => ({  
      path,  
      markdown: await htmlToMarkdown(html, request.mode || 'aggregate')  
    }))  
  );  
    
  // 新增文件保存功能  
  if (request.saveFormat) {  
    await saveToFiles(pages, request);  
  }  
    
  return {  
    content: pages.map(page => ({  
      type: 'text',  
      text: `# ${page.path}\n\n${page.markdown}`  
    })),  
    savedFiles: request.saveFormat ? await getSavedFilePaths(pages, request) : null  
  };  
}
文件保存功能实现：

async function saveToFiles(pages, request) {  
  const outputDir = request.outputPath || process.env.OUTPUT_DIR || './output';  
  await fs.mkdir(outputDir, { recursive: true });  
    
  for (const page of pages) {  
    const filename = sanitizeFilename(page.path);  
    const extension = request.saveFormat === 'txt' ? 'txt' : 'md';  
    const filepath = path.join(outputDir, `${filename}.${extension}`);  
      
    let content = page.markdown;  
    if (request.saveFormat === 'txt') {  
      // 移除Markdown格式，保留纯文本  
      content = stripMarkdown(content);  
    }  
      
    await fs.writeFile(filepath, content, 'utf8');  
  }  
}
4. 批量调用支持
支持VCP的串行调用语法，处理带数字后缀的参数：

function parseBatchRequest(request) {  
  const commands = [];  
  let index = 1;  
    
  while (request[`command${index}`]) {  
    const command = {  
      command: request[`command${index}`],  
      url: request[`url${index}`],  
      mode: request[`mode${index}`],  
      saveFormat: request[`saveFormat${index}`],  
      // ... 其他参数  
    };  
    commands.push(command);  
    index++;  
  }  
    
  return commands.length > 0 ? commands : [request];  
}
