// 文件保存功能伪代码  
function saveContent(pages: PageObject[], options: SaveOptions) {  
  for (const page of pages) {  
    const filename = `${sanitizeFilename(page.path)}.${options.format}`  
    const filepath = path.join(options.outputDir, filename)  
      
    if (options.format === 'md') {  
      writeFileSync(filepath, page.markdown)  
    } else if (options.format === 'txt') {  
      // 移除Markdown格式，保留纯文本  
      const plainText = stripMarkdown(page.markdown)  
      writeFileSync(filepath, plainText)  
    }  
  }  
}
